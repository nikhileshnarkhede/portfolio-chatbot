# =====================================================================
# portfolio_Chatbot - runtime dependencies
#
# Versions are PINNED with ==. This project exists to measure the effect
# of parameter changes, so the dependency set has to be a constant. An
# unpinned langchain-core minor bump can shift retrieval or prompt
# formatting and silently invalidate every stored result in runs/.
#
# Resolved 2026-08-25 against Python 3.12.
#
#   python -m venv .venv
#   .\.venv\Scripts\Activate.ps1
#   python -m pip install --upgrade pip
#   pip install -r requirements.txt
#   pip freeze > requirements.lock.txt     # full transitive lock, commit it
#
# Only DIRECT dependencies are listed here. requirements.lock.txt is the
# reproducible artifact; this file is the human-readable intent.
# =====================================================================

# ---- orchestration --------------------------------------------------
# langgraph: the graph runtime (state, nodes, edges, checkpointers).
# langchain-core: Documents, prompts, runnables, output parsers.
# The umbrella `langchain` package is deliberately NOT installed - nothing
# in this project imports it, and it drags in extra resolution constraints.
langgraph==1.2.11
langchain-core==1.6.0

# ---- LLM provider ---------------------------------------------------
# ChatGroq. Pulls the `groq` SDK (0.37.1) transitively.
langchain-groq==1.1.3

# ---- retrieval ------------------------------------------------------
# langchain-community: the FAISS vectorstore wrapper.
# langchain-huggingface: HuggingFaceEmbeddings.
# langchain-text-splitters: RecursiveCharacterTextSplitter, used only as
#   the oversize-chunk safeguard in ingestion/chunker.py.
langchain-community==0.4.2
langchain-huggingface==1.2.2
langchain-text-splitters==1.1.2

# ---- vector index + embeddings --------------------------------------
# sentence-transformers pulls torch (2.13.0) and transformers (5.15.1).
# On Windows the PyPI torch wheel is CPU-only, so no special index URL is
# needed - expect roughly 2-3 GB of install for this group regardless.
# embedding.device in configs/default.json is "cpu"; keep it that way
# unless you install a CUDA build deliberately.
faiss-cpu==1.15.0
sentence-transformers==6.0.0
torchvision
# ---- config ---------------------------------------------------------
# pydantic backs config.py's validation, freezing and extra="forbid".
# Already a transitive langchain-core dependency; pinned explicitly
# because config.py imports it directly.
pydantic==2.13.4

# ---- UI -------------------------------------------------------------
streamlit==1.62.0

# ---- environment ----------------------------------------------------
# Loads GROQ_API_KEY from .env for CLI entrypoints (scripts/ingest.py,
# scripts/run_eval.py). Streamlit reads .streamlit/secrets.toml instead.
python-dotenv==1.2.3
